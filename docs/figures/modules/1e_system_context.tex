% =============================================================================
% Figure 1E-1: Module 1E AWG functional architecture
% =============================================================================
% Maps the standard arbitrary-waveform-generator architecture (sample clock
% generator, address generator, waveform memory, DAC, reconstruction filter,
% signal conditioning) to the PMVB Module 1E realization. Each block carries
% its textbook function as the headline and the PMVB implementation
% (Pico firmware role, IC P/N, key parameters) in italic muted text below.
%
% Style follows the canonical Tektronix / Keysight AWG block diagram so a
% reader who knows AWGs can recognize the topology, then read the italic
% sub-labels to see how this implementation fills each role.
%
% Compile:  pdflatex 1e_system_context.tex
% Convert:  pdftocairo -svg 1e_system_context.pdf
% =============================================================================
\documentclass[border=8pt]{standalone}
%
\input{../style/pmvb-figures.sty}
%
\begin{document}
\begin{tikzpicture}[pmvb figure]

  % ---------------------------------------------------------------------
  % Top row: sample clock generator + external 10 MHz reference input
  % ---------------------------------------------------------------------
  \node[pmvb ic, minimum width=36mm, minimum height=16mm]
    (clk) at (1.6, 3.4) {};
  \node[font=\sffamily\bfseries\scriptsize, text=pmvbFg]
    at ([yshift=5mm]clk.center) {Sample clock generator};
  \node[font=\sffamily\scriptsize\itshape, text=pmvbMuted]
    at ([yshift=1mm]clk.center) {Pico PIO + DMA};
  \node[font=\sffamily\scriptsize\itshape, text=pmvbMuted]
    at ([yshift=-2mm]clk.center) {30 to 50 MSPS};
  \node[font=\sffamily\scriptsize\itshape, text=pmvbMuted]
    at ([yshift=-5mm]clk.center) {locks to ext.\ 10 MHz};

  % External reference input (enters from above the clock generator)
  \node[font=\sffamily\scriptsize, text=pmvbFg, anchor=south]
    (extref) at (1.6, 5.4) {Ext.\ 10\,MHz ref};
  \draw[pmvb digital] (extref.south) -- (clk.north);

  % ---------------------------------------------------------------------
  % Main signal-flow row, left to right
  % ---------------------------------------------------------------------

  % Trigger IN annotation
  \node[font=\sffamily\scriptsize, text=pmvbFg, anchor=east, align=right]
    (trigin) at (-0.4, 0) {Trigger IN\\(GP17)};

  % Address generator
  \node[pmvb ic, minimum width=30mm, minimum height=18mm]
    (addr) at (1.6, 0) {};
  \node[font=\sffamily\bfseries\scriptsize, text=pmvbFg]
    at ([yshift=6mm]addr.center) {Address};
  \node[font=\sffamily\bfseries\scriptsize, text=pmvbFg]
    at ([yshift=3mm]addr.center) {generator};
  \node[font=\sffamily\scriptsize\itshape, text=pmvbMuted]
    at ([yshift=-1mm]addr.center) {Pico firmware:};
  \node[font=\sffamily\scriptsize\itshape, text=pmvbMuted]
    at ([yshift=-4mm]addr.center) {DDS phase accum};
  \node[font=\sffamily\scriptsize\itshape, text=pmvbMuted]
    at ([yshift=-7mm]addr.center) {or DMA counter};

  % Waveform memory
  \node[pmvb ic, minimum width=34mm, minimum height=18mm]
    (mem) at (5.5, 0) {};
  \node[font=\sffamily\bfseries\scriptsize, text=pmvbFg]
    at ([yshift=6mm]mem.center) {Waveform memory};
  \node[font=\sffamily\scriptsize\itshape, text=pmvbMuted]
    at ([yshift=2mm]mem.center) {Pico SRAM up to};
  \node[font=\sffamily\scriptsize\itshape, text=pmvbMuted]
    at ([yshift=-1mm]mem.center) {{$\sim$}256\,K samples,};
  \node[font=\sffamily\scriptsize\itshape, text=pmvbMuted]
    at ([yshift=-4mm]mem.center) {or DDS on the fly,};
  \node[font=\sffamily\scriptsize\itshape, text=pmvbMuted]
    at ([yshift=-7mm]mem.center) {or 4\,MB flash stream};

  % DAC
  \node[pmvb ic, minimum width=26mm, minimum height=18mm]
    (dac) at (9.4, 0) {};
  \node[font=\sffamily\bfseries\scriptsize, text=pmvbFg]
    at ([yshift=6mm]dac.center) {DAC};
  \node[font=\sffamily\scriptsize\itshape, text=pmvbMuted]
    at ([yshift=2mm]dac.center) {AD9742};
  \node[font=\sffamily\scriptsize\itshape, text=pmvbMuted]
    at ([yshift=-1mm]dac.center) {12-bit};
  \node[font=\sffamily\scriptsize\itshape, text=pmvbMuted]
    at ([yshift=-4mm]dac.center) {parallel I-out};
  \node[font=\sffamily\scriptsize\itshape, text=pmvbMuted]
    at ([yshift=-7mm]dac.center) {30 to 50 MSPS};

  % Reconstruction filter
  \node[pmvb ic, minimum width=32mm, minimum height=18mm]
    (lpf) at (13.0, 0) {};
  \node[font=\sffamily\bfseries\scriptsize, text=pmvbFg]
    at ([yshift=6mm]lpf.center) {Reconstruction};
  \node[font=\sffamily\bfseries\scriptsize, text=pmvbFg]
    at ([yshift=3mm]lpf.center) {filter};
  \node[font=\sffamily\scriptsize\itshape, text=pmvbMuted]
    at ([yshift=-1mm]lpf.center) {5th-order Butterworth,};
  \node[font=\sffamily\scriptsize\itshape, text=pmvbMuted]
    at ([yshift=-4mm]lpf.center) {two per-leg L-C-L-C-L};
  \node[font=\sffamily\scriptsize\itshape, text=pmvbMuted]
    at ([yshift=-7mm]lpf.center) {ladders, $\sim$12\,MHz};

  % Signal conditioning
  \node[pmvb ic, minimum width=36mm, minimum height=18mm]
    (cond) at (17.0, 0) {};
  \node[font=\sffamily\bfseries\scriptsize, text=pmvbFg]
    at ([yshift=6mm]cond.center) {Signal conditioning};
  \node[font=\sffamily\scriptsize\itshape, text=pmvbMuted]
    at ([yshift=2mm]cond.center) {AD8056 diff-to-SE,};
  \node[font=\sffamily\scriptsize\itshape, text=pmvbMuted]
    at ([yshift=-1mm]cond.center) {gain 20, $\pm$10\,V;};
  \node[font=\sffamily\scriptsize\itshape, text=pmvbMuted]
    at ([yshift=-4mm]cond.center) {SP3T: 50\,$\Omega$ /};
  \node[font=\sffamily\scriptsize\itshape, text=pmvbMuted]
    at ([yshift=-7mm]cond.center) {high-Z / 10\,k$\Omega$};

  % BNC output
  \node[pmvb connector, right=8mm of cond.east] (bnc) {BNC};
  \node[pmvb annotation, below=0.5mm of bnc] {OUT};

  % ---------------------------------------------------------------------
  % Bottom row: microprocessor / SCPI interface
  % ---------------------------------------------------------------------
  \node[pmvb ic, minimum width=42mm, minimum height=16mm]
    (cpu) at (5.5, -3.7) {};
  \node[font=\sffamily\bfseries\scriptsize, text=pmvbFg]
    at ([yshift=5mm]cpu.center) {Microprocessor};
  \node[font=\sffamily\bfseries\scriptsize, text=pmvbFg]
    at ([yshift=2mm]cpu.center) {/ SCPI interface};
  \node[font=\sffamily\scriptsize\itshape, text=pmvbMuted]
    at ([yshift=-2mm]cpu.center) {Pico USB-TMC,};
  \node[font=\sffamily\scriptsize\itshape, text=pmvbMuted]
    at ([yshift=-5mm]cpu.center) {SCPI command parser};

  % USB-TMC into the microprocessor from below
  \node[font=\sffamily\scriptsize, text=pmvbFg, anchor=north, align=center]
    (usbtmc) at (5.5, -6.0) {USB-TMC\\to Pi 5 host};
  \draw[pmvb bus] (usbtmc.north) -- (cpu.south);

  % Sync OUT (GP16) from the microprocessor to the chassis trigger bus
  \node[font=\sffamily\scriptsize, text=pmvbFg, anchor=west, align=left]
    (syncout) at (11.4, -3.7) {Sync OUT (GP16)\\to chassis trigger bus};
  \draw[pmvb digital] (cpu.east) -- (syncout.west);

  % ---------------------------------------------------------------------
  % Signal-flow arrows
  % ---------------------------------------------------------------------

  % Trigger IN -> Address generator
  \draw[pmvb digital] (trigin) -- (addr.west);

  % Address -> Memory (control)
  \draw[pmvb digital] (addr.east) -- (mem.west);

  % Memory -> DAC (12-bit parallel data bus)
  \draw[pmvb bus] (mem.east) -- node[pmvb bus width]{$\times 12$} (dac.west);

  % DAC -> Filter (analog differential, shown single-ended for clarity)
  \draw[pmvb analog] (dac.east) -- (lpf.west);

  % Filter -> Conditioning
  \draw[pmvb analog] (lpf.east) -- (cond.west);

  % Conditioning -> BNC
  \draw[pmvb analog] (cond.east)
    -- node[above=0.5mm]{\pmvblabel{$\pm$10\,V}}
    (bnc.west);

  % Sample clock generator -> Address generator (CLK).
  % The DAC also receives this clock in the physical implementation
  % (its CLOCK pin latches data on each edge), but the architectural
  % diagram follows textbook convention: clock drives the address gen.
  \draw[pmvb digital] (clk.south)
    -- node[right=0.5mm]{\pmvblabel{CLK}}
    (addr.north);

  % Microprocessor -> Waveform memory (load samples + read status back via
  % the same path). Straight vertical line; label sits to the left of the
  % line to avoid overlapping the arrow shaft.
  \draw[pmvb digital] (cpu.north)
    -- node[midway, anchor=east, xshift=-2mm]
       {\pmvblabel{load samples / status}}
    (mem.south);

  % ---------------------------------------------------------------------
  % Bottom annotation
  % ---------------------------------------------------------------------
  \node[pmvb annotation, below=4mm of usbtmc]
    {Textbook AWG block diagram (Tektronix / Keysight style),
     mapped to the Module 1E realization.\\
     Bold labels are the standard architectural roles;
     italic muted text is the PMVB implementation that fills each role.\\
     The Pico fills three roles at once (address generator, waveform memory,
     microprocessor) using firmware on a single MCU.};

\end{tikzpicture}
\end{document}
