% =============================================================================
% Figure 1E-3: Module 1E typical application schematic
% =============================================================================
% Pico 2 W -> MCP4922 -> TL072 output buffer -> BNC, with the bypass network
% from the MCP4922 datasheet figure 6-1 (DS22250A section 6.2).
%
% This figure shows the SPI command path, the V_DD bypass network, and the
% TL072 output buffer at unity-gain configuration for visual simplicity.
% The full +/-10 V level-shift and gain network is described in the Module
% 1E design doc, section 3.
%
% Layout decisions:
%  - MCP4922 chip title placed at BOTTOM of the body so V_OUTA / V_OUTB pins
%    can sit higher and align horizontally with their op-amp + inputs.
%  - VOUTA at y=1.6 (matching bufa.+), VOUTB at y=-0.8 (matching bufb.+) so
%    the 0--3.3 V analog wires are STRAIGHT HORIZONTAL, not diagonal.
%  - Bypass network uses plain circuit wires (no colored arrows): +3.3 V text
%    label sits at the top, white wire descends through the cap junction to
%    the chip's V_DD pin, with V_DD labeled inside the chip body.
%  - Op-amp feedback routes via a T-junction past the triangle apex (on the
%    output wire), so the vertical-up segment is OUTSIDE the triangle.
%
% Compile:  pdflatex 1e_typical_app.tex
% Convert:  pdftocairo -svg 1e_typical_app.pdf
% =============================================================================
\documentclass[border=8pt]{standalone}
%
\input{../style/pmvb-figures.sty}
%
\begin{document}
\begin{circuitikz}[pmvb figure]
  %
  % =========================================================================
  % LEFT: Pico 2 W
  % =========================================================================
  \node[pmvb ic, minimum width=32mm, minimum height=44mm, anchor=west]
    (pico) at (0, 0) {};
  %
  \node[font=\sffamily\bfseries\small, text=pmvbFg]
    at ([yshift=-5mm]pico.north) {Pico 2 W};
  %
  \coordinate (pico_cs)   at ($(pico.east) + (0,  1.0)$);
  \coordinate (pico_sck)  at ($(pico.east) + (0,  0.3)$);
  \coordinate (pico_sdi)  at ($(pico.east) + (0, -0.3)$);
  \coordinate (pico_ldac) at ($(pico.east) + (0, -1.0)$);
  %
  \node[font=\sffamily\scriptsize, text=pmvbFg, anchor=east]
    at ($(pico_cs)   + (-2mm, 0)$) {GP17};
  \node[font=\sffamily\scriptsize, text=pmvbFg, anchor=east]
    at ($(pico_sck)  + (-2mm, 0)$) {GP18};
  \node[font=\sffamily\scriptsize, text=pmvbFg, anchor=east]
    at ($(pico_sdi)  + (-2mm, 0)$) {GP19};
  \node[font=\sffamily\scriptsize, text=pmvbFg, anchor=east]
    at ($(pico_ldac) + (-2mm, 0)$) {GP20};
  %
  % =========================================================================
  % MIDDLE: MCP4922 (title at BOTTOM of body)
  % =========================================================================
  \node[pmvb ic, minimum width=28mm, minimum height=44mm]
    at (7.5, 0) (dac) {};
  %
  % Title at the BOTTOM of the chip body (yshift=+5mm of south)
  \node[font=\sffamily\bfseries\small, text=pmvbFg]
    at ([yshift=5mm]dac.south) {MCP4922};
  %
  % MCP4922 pin coordinates.
  % VOUTA / VOUTB y values chosen to MATCH bufa.+ / bufb.+ (which are at
  % y=center+0.4 with yscale=-1) so the analog wires are horizontal.
  \coordinate (dac_cs)    at ($(dac.west)  + (0,  1.0)$);
  \coordinate (dac_sck)   at ($(dac.west)  + (0,  0.3)$);
  \coordinate (dac_sdi)   at ($(dac.west)  + (0, -0.3)$);
  \coordinate (dac_ldac)  at ($(dac.west)  + (0, -1.0)$);
  \coordinate (dac_vdd)   at (dac.north);
  \coordinate (dac_vss)   at (dac.south);
  \coordinate (dac_vouta) at ($(dac.east)  + (0,  1.6)$);
  \coordinate (dac_voutb) at ($(dac.east)  + (0, -0.8)$);
  %
  % Pin labels INSIDE the MCP4922 body
  \node[font=\sffamily\scriptsize, text=pmvbFg, anchor=west]
    at ($(dac_cs)   + (2mm, 0)$) {$\overline{\text{CS}}$};
  \node[font=\sffamily\scriptsize, text=pmvbFg, anchor=west]
    at ($(dac_sck)  + (2mm, 0)$) {SCK};
  \node[font=\sffamily\scriptsize, text=pmvbFg, anchor=west]
    at ($(dac_sdi)  + (2mm, 0)$) {SDI};
  \node[font=\sffamily\scriptsize, text=pmvbFg, anchor=west]
    at ($(dac_ldac) + (2mm, 0)$) {$\overline{\text{LDAC}}$};
  \node[font=\sffamily\scriptsize, text=pmvbFg, anchor=east]
    at ($(dac_vouta) + (-2mm, 0)$) {$V_{OUTA}$};
  \node[font=\sffamily\scriptsize, text=pmvbFg, anchor=east]
    at ($(dac_voutb) + (-2mm, 0)$) {$V_{OUTB}$};
  %
  % VDD pin label inside the chip body, hugging the top edge to keep clear
  % of the V_OUTA pin label below
  \node[font=\sffamily\scriptsize, text=pmvbFg, anchor=north]
    at ($(dac_vdd) + (0, -1mm)$) {$V_{DD}$};
  %
  % --- SPI wires Pico -> MCP4922 ---
  \draw[pmvb digital] (pico_cs)
    -- node[above=0.5mm]{\pmvblabel{$\overline{\text{CS}}$}}    (dac_cs);
  \draw[pmvb digital] (pico_sck)
    -- node[above=0.5mm]{\pmvblabel{SCK 20\,MHz}}              (dac_sck);
  \draw[pmvb digital] (pico_sdi)
    -- node[above=0.5mm]{\pmvblabel{SDI \textbullet{} 16-bit frame}} (dac_sdi);
  \draw[pmvb control] (pico_ldac)
    -- node[above=0.5mm]{\pmvblabel{$\overline{\text{LDAC}}$}} (dac_ldac);
  %
  % =========================================================================
  % BYPASS NETWORK above MCP4922 (all plain circuit wires, no colored arrows)
  % =========================================================================
  % "+3.3 V" red text label at the top
  \node[font=\sffamily\scriptsize, text=pmvbRed]
    at (7.5, 4.7) (vdd_label) {+3.3 V};
  %
  % Cap junction line and cap-top coordinates
  \coordinate (c1_top)  at (6.0, 4.0);
  \coordinate (vdd_jct) at (7.5, 4.0);
  \coordinate (c2_top)  at (9.0, 4.0);
  %
  % Plain circuit wire from "+3.3 V" label down through the junction to the
  % chip's V_DD pin. NO arrow head (this is a passive rail, not a signal).
  \draw[pmvbFg, line width=0.5pt]
    (vdd_label.south) -- (vdd_jct) -- (dac_vdd);
  %
  % Horizontal junction connecting the two cap tops
  \draw[pmvbFg, line width=0.5pt] (c1_top) -- (c2_top);
  %
  % Junction dot where the vertical wire meets the horizontal junction
  \fill[pmvbFg] (vdd_jct) circle (0.5mm);
  %
  % Bypass caps (dot at top only; ground glyph implies the bottom connection)
  \draw (c1_top) to[C, l_={\pmvblabel{C1 10\,$\mu$F}}, *-]
    ++(0, -1.0) node[ground]{};
  \draw (c2_top) to[C, l^={\pmvblabel{C2 0.1\,$\mu$F}}, *-]
    ++(0, -1.0) node[ground]{};
  %
  % VSS to ground (small stub below chip)
  \draw (dac_vss) to[short, *-] ++(0, -0.5) node[ground]{};
  %
  % =========================================================================
  % RIGHT: MCP6232 unity-gain buffers, with horizontal VOUT wires
  % =========================================================================
  % bufa: center y = VOUTA.y - 0.4 = 1.2 (so bufa.+ aligns with VOUTA at y=1.6)
  % bufb: center y = VOUTB.y - 0.4 = -1.2 (so bufb.+ aligns with VOUTB at y=-0.8)
  \node[op amp, scale=0.8, yscale=-1] (bufa) at (15, 1.2) {};
  \node[op amp, scale=0.8, yscale=-1] (bufb) at (15, -1.2) {};
  %
  % --- VOUTA -> bufa.+ (STRAIGHT horizontal at y=1.6) ---
  \draw[pmvb analog] (dac_vouta) --
    node[above=0.5mm]{\pmvblabel{0--3.3\,V analog}}
    (bufa.+);
  %
  % --- VOUTB -> bufb.+ (STRAIGHT horizontal at y=-0.8) ---
  \draw[pmvb analog] (dac_voutb) --
    node[above=0.5mm]{\pmvblabel{0--3.3\,V analog}}
    (bufb.+);
  %
  % --- BNC output connectors ---
  \node[pmvb connector, right=14mm of bufa.out] (bnc_a) {OUT\\A};
  \node[pmvb connector, right=14mm of bufb.out] (bnc_b) {OUT\\B};
  %
  % --- Output wires from op-amp .out to BNC, with feedback T-junctions ---
  % Tap point on the output wire, sized so the feedback's vertical-up
  % segment is clearly past the triangle apex (5mm clearance).
  \coordinate (bufa_tap) at ($(bufa.out) + (5mm, 0)$);
  \coordinate (bufb_tap) at ($(bufb.out) + (5mm, 0)$);
  %
  \draw[pmvb analog] (bufa.out) -- (bnc_a.west);
  \draw[pmvb analog] (bufb.out) -- (bnc_b.west);
  %
  % --- Feedback wires: from .- DOWN well past triangle bottom edge,
  % then -| to a tap point past the apex. Larger 7mm down + 5mm right
  % offsets give visible clearance from the triangle outline.
  \draw (bufa.-) -- ++(0, -7mm) -| (bufa_tap);
  \draw (bufb.-) -- ++(0, -7mm) -| (bufb_tap);
  %
  % Junction dots at the tap points
  \fill[pmvbFg] (bufa_tap) circle (0.5mm);
  \fill[pmvbFg] (bufb_tap) circle (0.5mm);
  %
  % =========================================================================
  % Bottom annotation (caption is rendered in the design doc, not in SVG)
  % =========================================================================
  \node[pmvb annotation, below=14mm of dac.south]
    {C1 = 10\,$\mu$F tantalum (bulk decoupling)\\
     C2 = 0.1\,$\mu$F ceramic (HF decoupling)\\
     Place both within 4\,mm of $V_{DD}$ pin (DS22250A \S6.2)};
\end{circuitikz}
\end{document}
